cmake_minimum_required(VERSION 3.14)
project(apexis_usb_io LANGUAGES CXX)
add_library(apexis_usb_io SHARED usb_io.cpp)
target_compile_features(apexis_usb_io PRIVATE cxx_std_17)
target_link_libraries(apexis_usb_io PRIVATE winusb setupapi)
target_compile_definitions(apexis_usb_io PRIVATE UNICODE _UNICODE NOMINMAX)
if(MSVC)
  target_compile_options(apexis_usb_io PRIVATE /W4 /WX /EHsc)
endif()
option(APEXIS_USB_TESTS "Build hardware-free Windows I/O regression" OFF)
if(APEXIS_USB_TESTS)
  add_executable(apexis_usb_io_test usb_io_test.cpp)
  target_compile_features(apexis_usb_io_test PRIVATE cxx_std_17)
  target_compile_definitions(apexis_usb_io_test PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(apexis_usb_io_test PRIVATE winusb setupapi)
  if(MSVC)
    target_compile_options(apexis_usb_io_test PRIVATE /W4 /WX /EHsc)
  endif()
  enable_testing()
  add_test(NAME usb_io COMMAND apexis_usb_io_test)
  set_tests_properties(usb_io PROPERTIES TIMEOUT 15)
endif()
