cmake_minimum_required(VERSION 3.16)
project(apexis_macos_usb LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "This target requires macOS and the Apple SDK")
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(apexis_macos_usb SHARED ../Runner/usb_io.cpp)
target_link_libraries(apexis_macos_usb PRIVATE "-framework IOKit" "-framework CoreFoundation")
target_compile_options(apexis_macos_usb PRIVATE -Wall -Wextra)
enable_testing()
add_executable(macos_usb_test usb_io_test.cpp)
target_link_libraries(macos_usb_test PRIVATE "-framework IOKit" "-framework CoreFoundation")
target_compile_options(macos_usb_test PRIVATE -Wall -Wextra)
add_test(NAME macos_usb_contract COMMAND macos_usb_test)
